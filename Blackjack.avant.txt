function Counter({ value }: { value: number }) {
  const [shown, setShown] = useState(value);
  const from = useRef(value);
  useEffect(() => {
    const start = performance.now();
    const a = from.current;
    let raf = 0;
    const tick = (t: number) => {
      const p = Math.min(1, (t - start) / 600);
      const v = Math.round(a + (value - a) * (1 - Math.pow(1 - p, 3)));
      from.current = v;
      setShown(v);
      if (p < 1) raf = requestAnimationFrame(tick);
    };
    raf = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(raf);
  }, [value]);
  return <>{money(shown)}</>;
}

function ChipStack({ amount }: { amount: number }) {
  if (amount <= 0) return null;
  const stacks: { v: number; n: number }[] = [];
  let rest = amount;
  for (const v of [...CHIPS].reverse()) {
    const n = Math.floor(rest / v);
    if (n > 0) {
      stacks.push({ v, n: Math.min(n, 8) });
      rest -= n * v;
    }
  }
  return (
    <div key={amount} className="animate-pop flex items-end justify-center gap-1.5" aria-label={`${amount} chips bet`}>
      {stacks.map(({ v, n }) => (
        <div key={v} className="relative w-9" style={{ height: 14 + (n - 1) * 5 }}>
          {Array.from({ length: n }).map((_, k) => (
            <div
              key={k}
              className={`absolute left-0 h-3.5 w-9 rounded-full border-2 border-dashed border-white/70 shadow ${CHIP_BG[v]}`}
              style={{ bottom: k * 5 }}
            />
          ))}
        </div>
      ))}
    </div>
  );
}

function Spark({ data }: { data: number[] }) {
  if (data.length < 2) return <p className="text-xs text-white/30">Play a few hands to see your trend.</p>;
  const w = 240, h = 48;
  const min = Math.min(...data), max = Math.max(...data);
  const r = max - min || 1;
  const pts = data.map((v, i) => `${(i / (data.length - 1)) * w},${h - 4 - ((v - min) / r) * (h - 8)}`).join(" ");
  return (
    <svg viewBox={`0 0 ${w} ${h}`} className="w-full" role="img" aria-label="Chip balance over this session">
      <polyline points={pts} fill="none" stroke="#d7b45b" strokeWidth="2" strokeLinejoin="round" />
    </svg>
  );
}

/* ---------- Jeu ---------- */

export default function Blackjack() {
  // Un seul état + une ref miroir : les fonctions async lisent toujours la valeur à jour.
  const ref = useRef<S>(INITIAL);
  const [s, setS] = useState<S>(INITIAL);
  const [showHint, setShowHint] = useState(false);
  const [hydrated, setHydrated] = useState(false);
  const [showCount, setShowCount] = useState(false);

  const set = (patch: Partial<S>) => {
    ref.current = { ...ref.current, ...patch };
    setS(ref.current);
  };
  const setHand = (i: number, patch: Partial<Hand>) =>
    set({ hands: ref.current.hands.map((h, k) => (k === i ? { ...h, ...patch } : h)) });

  useEffect(() => {
    try {
      const raw = localStorage.getItem("royalace:blackjack");
      if (raw) {
        const d = JSON.parse(raw);
        if (typeof d.balance === "number") {
          const settings = { ...DEFAULT_SETTINGS, ...d.settings };
          setMuted(settings.muted);
          set({
            balance: d.balance,
            stats: { ...EMPTY_STATS, ...d.stats },
            settings,
            trend: [d.balance],
            unlocked: Array.isArray(d.unlocked) ? d.unlocked : [],
            records: { ...EMPTY_RECORDS, ...d.records },
            bet: Math.max(0, Math.min(ref.current.bet, floor10(d.balance))),
          });
        }
      }
    } catch {}
    setHydrated(true);
  }, []);

  useEffect(() => {
    if (!hydrated) return;
    try {
      localStorage.setItem(
        "royalace:blackjack",
        JSON.stringify({ balance: s.balance, stats: s.stats, settings: s.settings, unlocked: s.unlocked, records: s.records })
      );
    } catch {}
  }, [hydrated, s.balance, s.stats, s.settings, s.unlocked, s.records]);

  // ---------- Pioche ----------
  function draw(): Card {
    let shoe = ref.current.shoe;
    if (shoe.length === 0) shoe = makeShoe(ref.current.settings.decks);
    const card = shoe[shoe.length - 1];
    set({ shoe: shoe.slice(0, -1) });
    return card;
  }
  function giveCard(i: number) {
    const card = draw();
    sfx.deal();
    setHand(i, { cards: [...ref.current.hands[i].cards, card] });
  }
  function giveDealer() {
    const card = draw();
    sfx.deal();
    set({ dealer: [...ref.current.dealer, card] });
  }
  function reveal() {
    if (!ref.current.hole) {
      set({ hole: true });
      sfx.flip();
    }
  }

  // ---------- Mise ----------
  const canBet = s.phase === "betting" || s.phase === "settled";
  function addChip(v: number) {
    const c = ref.current;
    if (c.phase !== "betting" && c.phase !== "settled") return;
    if ((c.bet + v) * c.settings.spots + c.ppBet + c.t3Bet <= c.balance) {
      set({ bet: c.bet + v });
      sfx.chip();
    }
  }
  function setBetTo(amount: number) {
    const c = ref.current;
    if (c.phase !== "betting" && c.phase !== "settled") return;
    const room = c.balance - c.ppBet - c.t3Bet;
    set({ bet: Math.max(0, Math.min(floor10(amount), floor10(room / c.settings.spots))) });
    sfx.chip();
  }
  function updateSettings(patch: Partial<Settings>) {
    const c = ref.current;
    if (patch.muted !== undefined) setMuted(patch.muted);
    else if (patch.theme === undefined && c.phase !== "betting" && c.phase !== "settled") return;
    const settings = { ...c.settings, ...patch };
    set({
      settings,
      ...(patch.decks ? { shoe: [] } : {}),
      ...(patch.spots
        ? { bet: Math.min(c.bet, floor10((c.balance - c.ppBet - c.t3Bet) / patch.spots)) }
        : {}),
    });
  }

  function adjustSide(key: "ppBet" | "t3Bet", delta: number) {
    const c = ref.current;
    if (c.phase !== "betting" && c.phase !== "settled") return;
    const next = c[key] + delta;
    if (next < 0 || next > 100) return;
    if (c.bet * c.settings.spots + c.ppBet + c.t3Bet + delta > c.balance) return;
    set({ [key]: next } as Partial<S>);
    sfx.chip();
  }

  // ---------- Déroulé d'une manche ----------
  function unlock(id: string) {
    const c = ref.current;
    if (c.unlocked.includes(id)) return;
    const a = ACHIEVEMENTS.find((x) => x.id === id);
    set({ unlocked: [...c.unlocked, id], toast: `Achievement unlocked: ${a?.title ?? id}` });
    sfx.win();
    window.setTimeout(() => set({ toast: null }), 3800);
  }

  /** Note ta décision avant qu'elle ne s'exécute et la compare à la stratégie de base. */
  function grade(action: Advice) {
    const c = ref.current;
    const h = c.hands[c.active];
    const up = c.dealer[0];
    if (!h || !up) return;
    const cd = h.cards.length === 2 && c.balance >= h.bet;
    const cs = canSplitCards(h.cards) && c.hands.length < 4 && c.balance >= h.bet;
    const best = advise(h.cards, up, cd, cs);
    const ok = best === action;
    set({
      stats: { ...c.stats, decisions: c.stats.decisions + 1, correct: c.stats.correct + (ok ? 1 : 0) },
      goodStreak: ok ? c.goodStreak + 1 : 0,
      coach: ok ? `✓ ${action} was the right play.` : `✗ You chose ${action}. Basic strategy says ${best}.`,
    });
    if (ok && c.goodStreak + 1 >= 10) unlock("sharp");
  }

  /** Paris annexes : résolus dès la distribution, sur la 1re main. */
  function resolveSide() {
    const c = ref.current;
    if (c.ppBet + c.t3Bet === 0) return;
    const p = c.hands[0].cards;
    let payout = 0;
    let big = false;
    const notes: string[] = [];
    if (c.ppBet > 0) {
      const w = perfectPairs(p[0], p[1]);
      if (w) {
        payout += c.ppBet * (w.odds + 1);
        notes.push(`${w.label} +${c.ppBet * w.odds}`);
        unlock("pair");
        if (w.odds >= 25) big = true;
      }
    }
    if (c.t3Bet > 0) {
      const w = twentyOnePlusThree(p[0], p[1], c.dealer[0]);
      if (w) {
        payout += c.t3Bet * (w.odds + 1);
        notes.push(`${w.label} +${c.t3Bet * w.odds}`);
        unlock("suited");
        if (w.odds >= 40) big = true;
      }
    }
    const sideNet = payout - c.ppBet - c.t3Bet;
    const now = ref.current;
    if (payout > 0) sfx.win();
    set({
      balance: now.balance + payout,
      sideNote: notes.length ? `Side bets: ${notes.join(" · ")}` : "Side bets lost.",
      stats: { ...now.stats, profit: now.stats.profit + sideNet },
      ...(big ? { celebrate: Date.now() } : {}),
    });
  }

  async function deal() {
    const c = ref.current;
    const { spots, decks } = c.settings;
    const stake = c.bet * spots + c.ppBet + c.t3Bet;
    if ((c.phase !== "betting" && c.phase !== "settled") || c.bet < MIN_BET || stake > c.balance) return;

    const fresh = c.shoe.length < Math.max(26, Math.floor(decks * 52 * 0.25));
    sfx.chip();
    set({
      shoe: fresh ? makeShoe(decks) : c.shoe,
      balance: c.balance - stake,
      lastBet: c.bet,
      hands: Array.from({ length: spots }, () => ({ cards: [], bet: c.bet, done: false, doubled: false, split: false })),
      dealer: [],
      hole: false,
      active: 0,
      insurance: 0,
      net: null,
      sideNote: null,
      celebrate: 0,
      phase: "dealing",
      shuffling: fresh,
      message: fresh ? "Shuffling a new shoe…" : "Dealing…",
    });
    if (fresh) {
      sfx.flip();
      await sleep(1100);
      set({ shuffling: false, message: "Dealing…" });
    }
    await sleep(250);

    for (let round = 0; round < 2; round++) {
      for (let i = 0; i < spots; i++) {
        giveCard(i);
        await sleep(380);
      }
      giveDealer();
      await sleep(380);
    }
    await sleep(100);
    resolveSide();

    const { dealer, hands, balance } = ref.current;
    const total = hands.reduce((a, h) => a + h.bet, 0);
    if (dealer[0].rank === "A" && balance >= total / 2) {
      set({ phase: "insurance", message: "Dealer shows an Ace. Take insurance?" });
      return;
    }
    await peek();
  }

  async function takeInsurance(take: boolean) {
    const c = ref.current;
    if (c.phase !== "insurance" || c.busy) return;
    set({ busy: true });
    if (take) {
      const cost = c.hands.reduce((a, h) => a + h.bet, 0) / 2;
      set({ balance: c.balance - cost, insurance: cost });
      sfx.chip();
    }
    await peek();
    set({ busy: false });
  }

  /** Le croupier vérifie s'il a un blackjack. Les blackjacks du joueur sont réglés tout de suite. */
  async function peek() {
    const { dealer, hands } = ref.current;
    const naturals = hands.map((h) => isBlackjack(h.cards));
    if (isBlackjack(dealer) || naturals.every(Boolean)) {
      set({ phase: "dealer" });
      reveal();
      await sleep(900);
      await settle();
      return;
    }
    set({ hands: hands.map((h, i) => (naturals[i] ? { ...h, done: true } : h)), phase: "player" });
    await advance();
  }

  async function advance() {
    const { hands } = ref.current;
    const next = hands.findIndex((h) => !h.done);
    if (next === -1) {
      await dealerPlay();
      return;
    }
    set({ active: next, message: hands.length > 1 ? `Hand ${next + 1}: your move.` : "Your move." });
  }

  async function hit() {
    const c = ref.current;
    if (c.phase !== "player" || c.busy) return;
    const i = c.active;
    grade("Hit");
    set({ busy: true });
    giveCard(i);
    await sleep(500);
    if (handValue(ref.current.hands[i].cards).total >= 21) {
      setHand(i, { done: true });
      await advance();
    }
    set({ busy: false });
  }

  async function stand() {
    const c = ref.current;
    if (c.phase !== "player" || c.busy) return;
    grade("Stand");
    set({ busy: true });
    setHand(c.active, { done: true });
    await advance();
    set({ busy: false });
  }

  async function double() {
    const c = ref.current;
    const h = c.hands[c.active];
    if (c.phase !== "player" || c.busy || !h || h.cards.length !== 2 || c.balance < h.bet) return;
    grade("Double");
    set({ busy: true, balance: c.balance - h.bet });
    sfx.chip();
    setHand(c.active, { bet: h.bet * 2, doubled: true });
    giveCard(c.active);
    await sleep(700);
    setHand(c.active, { done: true });
    await advance();
    set({ busy: false });
  }

  async function split() {
    const c = ref.current;
    const h = c.hands[c.active];
    if (c.phase !== "player" || c.busy || !h || !canSplitCards(h.cards) || c.hands.length >= 4 || c.balance < h.bet) return;
    const i = c.active;
    const aces = h.cards[0].rank === "A";
    grade("Split");
    set({ busy: true, balance: c.balance - h.bet });
    sfx.chip();

    const a: Hand = { ...h, cards: [h.cards[0]], split: true };
    const b: Hand = { cards: [h.cards[1]], bet: h.bet, done: false, doubled: false, split: true };
    const hands = [...c.hands];
    hands.splice(i, 1, a, b);
    set({ hands, message: "Split!" });
    await sleep(350);

    giveCard(i); await sleep(450);
    giveCard(i + 1); await sleep(450);

    // As splittés : une seule carte chacun. 21 : main terminée.
    set({
      hands: ref.current.hands.map((x, k) =>
        (k === i || k === i + 1) && (aces || handValue(x.cards).total === 21) ? { ...x, done: true } : x
      ),
    });
    await advance();
    set({ busy: false });
  }

  /** Abandon : seulement en début de main, sans split. Rend la moitié de la mise. */
  async function surrender() {
    const c = ref.current;
    const h = c.hands[0];
    if (c.phase !== "player" || c.busy || c.hands.length !== 1 || !h || h.cards.length !== 2 || h.split) return;
    set({ busy: true, phase: "dealer", message: "You surrender." });
    setHand(0, { done: true, result: "surrender" });
    reveal();
    await sleep(800);
    await settle();
    set({ busy: false });
  }

  async function dealerPlay() {
    set({ phase: "dealer", message: "Dealer's turn…" });
    reveal();
    await sleep(900);
    const { h17 } = ref.current.settings;
    const anyLive = ref.current.hands.some(
      (h) => h.result !== "surrender" && !(!h.split && isBlackjack(h.cards)) && handValue(h.cards).total <= 21
    );
    for (;;) {
      const { total, soft } = handValue(ref.current.dealer);
      if (!anyLive || total > 17 || (total === 17 && !(h17 && soft))) break;
      giveDealer();
      await sleep(800);
    }
    await settle();
  }

  async function settle() {
    const c = ref.current;
    const dv = handValue(c.dealer).total;
    const dBJ = isBlackjack(c.dealer);
    let payout = 0;
    let wagered = c.insurance;
    let gotBJ = false;
    let doubledWin = false;
    let splitWin = false;
    let won = 0, lost = 0, pushed = 0;

    const hands = c.hands.map((h) => {
      wagered += h.bet;
      if (h.result === "surrender") {
        payout += h.bet / 2;
        lost++;
        return h;
      }
      const pv = handValue(h.cards).total;
      const pBJ = !h.split && isBlackjack(h.cards);
      let result: Result;
      if (pv > 21) result = "bust";
      else if (pBJ && !dBJ) {
        result = "blackjack";
        payout += h.bet + (c.settings.bj65 ? (h.bet * 6) / 5 : (h.bet * 3) / 2);
        gotBJ = true;
      } else if (dBJ && !pBJ) result = "lose";
      else if (pBJ && dBJ) { result = "push"; payout += h.bet; }
      else if (dv > 21 || pv > dv) { result = "win"; payout += h.bet * 2; }
      else if (pv === dv) { result = "push"; payout += h.bet; }
      else result = "lose";

      if (result === "win" || result === "blackjack") {
        won++;
        if (h.doubled) doubledWin = true;
        if (h.split) splitWin = true;
      } else if (result === "push") pushed++;
      else lost++;
      return { ...h, result, done: true };
    });

    if (dBJ) payout += c.insurance * 3; // assurance payée 2:1
    const net = payout - wagered;
    const balance = c.balance + payout;
    const streak = net > 0 ? c.streak + 1 : net < 0 ? 0 : c.streak;
    const records: Records = {
      biggestWin: Math.max(c.records.biggestWin, net),
      bestStreak: Math.max(c.records.bestStreak, streak),
      peakBalance: Math.max(c.records.peakBalance, balance),
    };

    const message = hands.every((h) => h.result === "surrender")
      ? "You surrendered. Half your bet is returned."
      : gotBJ
      ? `Blackjack! Pays ${c.settings.bj65 ? "6 to 5" : "3 to 2"}.`
      : dBJ
      ? "Dealer has blackjack."
      : hands.every((h) => h.result === "bust")
      ? "Bust. The dealer wins."
      : net > 0
      ? "You win!"
      : net < 0
      ? "The dealer wins."
      : "Push. Your bet is returned.";

    if (gotBJ) sfx.blackjack();
    else if (net > 0) sfx.win();
    else if (hands.every((h) => h.result === "bust")) sfx.bust();
    else if (net < 0) sfx.lose();
    else sfx.push();

    // Les mises sont ramenées à ce que le solde permet encore
    const spots = c.settings.spots;
    const bet = Math.min(c.bet, floor10(balance / spots));
    const ppBet = Math.min(c.ppBet, floor10(balance - bet * spots));
    const t3Bet = Math.min(c.t3Bet, floor10(balance - bet * spots - ppBet));

    set({
      hands,
      balance,
      net,
      message,
      phase: "settled",
      hole: true,
      bet,
      ppBet,
      t3Bet,
      streak,
      records,
      celebrate: gotBJ || net >= 200 ? Date.now() : ref.current.celebrate,
      history: [...c.history, { net, bj: gotBJ }].slice(-14),
      trend: [...c.trend, balance].slice(-40),
      stats: {
        ...ref.current.stats,
        played: c.stats.played + 1,
        won: c.stats.won + won,
        lost: c.stats.lost + lost,
        pushed: c.stats.pushed + pushed,
        blackjacks: c.stats.blackjacks + (gotBJ ? 1 : 0),
        profit: ref.current.stats.profit + net,
      },
    });

    if (gotBJ) unlock("bj");
    if (streak >= 5) unlock("streak5");
    if (doubledWin) unlock("double");
    if (splitWin) unlock("split");
    if (balance >= 10000) unlock("rich");
  }

  function refill() {
    const c = ref.current;
    if ((c.phase !== "betting" && c.phase !== "settled") || c.balance >= MIN_BET) return;
    sfx.win();
    set({ balance: c.balance + REFILL, bet: 20, message: `Refill: ◈ ${REFILL} added. Good luck!` });
  }

  function resetAll() {
    const c = ref.current;
    if (c.phase !== "betting" && c.phase !== "settled") return;
    set({
      ...INITIAL,
      settings: c.settings,
      shoe: c.shoe,
      unlocked: c.unlocked,
      records: c.records,
      message: "Fresh start. Place your bet.",
    });
  }

  // ---------- Raccourcis clavier ----------
  const actions = useRef({ hit, stand, double, split, surrender, deal, addChip });
  actions.current = { hit, stand, double, split, surrender, deal, addChip };
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (e.metaKey || e.ctrlKey || e.altKey) return;
      const k = e.key.toLowerCase();
      const a = actions.current;
      if (k === "h") a.hit();
      else if (k === "s") a.stand();
      else if (k === "d") a.double();
      else if (k === "p") a.split();
      else if (k === "r") a.surrender();
      else if (k >= "1" && k <= "5") a.addChip(CHIPS[Number(k) - 1]);
      else if (k === " " || k === "enter") {
        const ph = ref.current.phase;
        if (ph === "betting" || ph === "settled") {
          e.preventDefault();
          a.deal();
        }
      }
    };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, []);

  // ---------- Dérivés d'affichage ----------
  const hand = s.hands[s.active];
  const playing = s.phase === "player" && !s.busy;
  const canDouble = playing && !!hand && hand.cards.length === 2 && s.balance >= hand.bet;
  const canSplit = playing && !!hand && canSplitCards(hand.cards) && s.hands.length < 4 && s.balance >= hand.bet;
  const canSurrender = playing && s.hands.length === 1 && !!hand && hand.cards.length === 2 && !hand.split;
  const advice =
    showHint && playing && hand && s.dealer[0] ? advise(hand.cards, s.dealer[0], canDouble, canSplit) : null;

  const dealerShown = s.hole ? s.dealer : s.dealer.slice(0, 1);
  const dealerVal = handValue(dealerShown);
  const outOfChips = canBet && s.balance < MIN_BET;
  const totalCards = s.settings.decks * 52;
  const shoePct = s.shoe.length ? Math.min(100, Math.round((s.shoe.length / totalCards) * 100)) : 100;
  const winRate = s.stats.played ? Math.round((s.stats.won / s.stats.played) * 100) : 0;
  const compact = s.hands.length > 1;
  const spots = s.settings.spots;
  const sides = s.ppBet + s.t3Bet;
  const stake = s.bet * spots + sides;
  const back = BACKS[s.settings.theme];
  const pileAmount = canBet ? s.bet * spots : s.hands.reduce((sum, h) => sum + h.bet, 0);

  // Comptage Hi-Lo : le sabot est équilibré, donc compte courant = -(somme du sabot restant),
  // sans compter la carte cachée du croupier tant qu'elle n'est pas retournée.
  const hiddenCard = !s.hole && s.dealer[1] ? hiLo(s.dealer[1]) : 0;
  const running = s.shoe.length ? -s.shoe.reduce((a, c) => a + hiLo(c), 0) - hiddenCard : 0;
  const decksLeft = s.shoe.length / 52;
  const trueCount = decksLeft > 0.5 ? running / decksLeft : running;

  const btn =
    "rounded-full px-6 py-3 text-sm font-bold outline-none transition focus-visible:ring-2 focus-visible:ring-[#e5c779] disabled:cursor-not-allowed disabled:opacity-30";
  const small =
    "rounded-full border border-white/10 py-2 text-xs text-white/60 outline-none transition hover:border-[#d7b45b] hover:text-[#e5c779] focus-visible:ring-2 focus-visible:ring-[#e5c779] disabled:opacity-30";

  return (
    <div className="grid gap-6 lg:grid-cols-[1fr_300px]">
      {/* ================= TABLE ================= */}
      <section className="overflow-clip rounded-3xl border border-[#c8a65b]/20 bg-[#101713] shadow-2xl">
        <div className="flex items-center justify-between border-b border-white/[0.07] px-6 py-4">
          <div>
            <h3 className="text-xl font-bold">Blackjack</h3>
            <p className="mt-0.5 text-xs text-white/40">
              {s.settings.decks} decks · dealer {s.settings.h17 ? "hits" : "stands on"} soft 17 · blackjack pays{" "}
              {s.settings.bj65 ? "6 to 5" : "3 to 2"}
            </p>
          </div>
          <div className="w-28 text-right text-[11px] text-white/40">
            Shoe · {s.shoe.length || totalCards} cards
            <div className="mt-1 h-1.5 overflow-hidden rounded-full bg-white/10">
              <div className="h-full rounded-full bg-[#d7b45b]/70 transition-all duration-500" style={{ width: `${shoePct}%` }} />
            </div>
          </div>
        </div>

        <div className={`relative ${THEMES[s.settings.theme]} px-3 py-8 sm:px-9`}>
          {s.celebrate > 0 && <Confetti key={s.celebrate} />}
          {s.shuffling && (
            <div className="absolute inset-0 z-20 flex flex-col items-center justify-center gap-3 bg-black/60" role="status">
              <div className="flex gap-3 text-3xl text-[#e5c779]">
                {["♠", "♥", "♦", "♣"].map((g, i) => (
                  <span key={g} className="animate-bounce" style={{ animationDelay: `${i * 120}ms` }}>{g}</span>
                ))}
              </div>
              <p className="text-sm text-white/80">Shuffling the shoe…</p>
            </div>
          )}
          {/* Croupier */}
          <div className="flex items-center justify-center gap-3 text-sm text-white/60">
            Dealer
            {s.dealer.length > 0 && (
              <span className="rounded-full bg-black/30 px-3 py-0.5 text-xs font-bold text-white">
                {s.hole && isBlackjack(s.dealer)
                  ? "Blackjack"
                  : dealerVal.soft && dealerVal.total < 21 && s.hole
                  ? `${dealerVal.total - 10}/${dealerVal.total}`
                  : dealerVal.total}
                {!s.hole && " + ?"}
              </span>
            )}
          </div>
          <div className="mt-4 flex min-h-28 justify-center sm:min-h-36">
            {s.dealer.map((card, i) => (
              <PlayingCard key={`d${i}-${card.rank}${card.suit}`} card={card} faceDown={i === 1 && !s.hole} overlap={i > 0} back={back} />
            ))}
          </div>

          {/* Message (annoncé aux lecteurs d'écran) */}
          <div className="my-7 flex flex-col items-center gap-2" role="status" aria-live="polite">
            <div key={s.message} className="animate-pop rounded-full border border-[#d7b45b]/30 bg-black/30 px-5 py-2 text-center text-sm text-[#e5c779]">
              {s.message}
            </div>
            {s.net !== null && s.net !== 0 && (
              <div className={`animate-pop text-2xl font-black ${s.net > 0 ? "text-[#f5d778]" : "text-red-300"}`}>
                {s.net > 0 ? "+" : "−"}◈ {money(Math.abs(s.net))}
              </div>
            )}
            {s.sideNote && (
              <div className={`animate-pop rounded-full border px-4 py-1 text-xs ${s.sideNote.includes("+") ? "border-[#e5c779]/60 bg-[#e5c779]/15 text-[#f5d778]" : "border-white/15 text-white/50"}`}>
                {s.sideNote}
              </div>
            )}
            {advice && (
              <div className="rounded-full border border-sky-300/40 bg-sky-400/10 px-4 py-1 text-xs text-sky-200">
                Basic strategy: {advice}
              </div>
            )}
          </div>

          {/* Joueur */}
          <div className="flex flex-wrap items-start justify-center gap-3 sm:gap-5">
            {s.hands.length === 0 && (
              <div className="flex min-h-28 items-center text-sm text-white/40 sm:min-h-36">Your cards will appear here</div>
            )}
            {s.hands.map((h, i) => {
              const v = handValue(h.cards);
              const isActive = s.phase === "player" && i === s.active && s.hands.length > 1;
              return (
                <div
                  key={i}
                  className={`rounded-2xl border p-2 transition sm:p-3 ${
                    isActive ? "border-[#e5c779]/70 bg-[#e5c779]/10" : "border-transparent"
                  } ${compact && !isActive && s.phase === "player" ? "opacity-75" : ""}`}
                >
                  <div className="mb-3 flex flex-wrap items-center justify-center gap-2 text-xs">
                    <span className="rounded-full bg-black/30 px-3 py-0.5 font-bold text-white">
                      {v.soft && v.total < 21 ? `${v.total - 10}/${v.total}` : v.total}
                    </span>
                    <span className="text-white/50">
                      ◈ {money(h.bet)}
                      {h.doubled && " (doubled)"}
                    </span>
                    {h.result && (
                      <span className={`animate-pop rounded-full border px-2.5 py-0.5 font-bold ${RESULT_UI[h.result].cls}`}>
                        {RESULT_UI[h.result].label}
                      </span>
                    )}
                  </div>
                  <div className={`flex justify-center ${compact ? "min-h-[88px] sm:min-h-28" : "min-h-28 sm:min-h-36"}`}>
                    {h.cards.map((card, k) => (
                      <PlayingCard key={`p${i}-${k}-${card.rank}${card.suit}`} card={card} overlap={k > 0} compact={compact} back={back} />
                    ))}
                  </div>
                </div>
              );
            })}
          </div>

          {/* Pile de jetons */}
          <div className="mt-6 flex min-h-12 flex-col items-center justify-end gap-1">
            <ChipStack amount={pileAmount} />
            {pileAmount > 0 && <span className="text-[11px] text-white/50">◈ {money(pileAmount)}</span>}
          </div>
        </div>

        {/* Contrôles */}
        <div className="sticky bottom-0 z-20 border-t border-white/10 bg-[#101713]/95 p-4 backdrop-blur sm:p-7 lg:static lg:border-0 lg:bg-transparent">
          {s.phase === "insurance" ? (
            <div className="flex flex-wrap justify-center gap-3">
              <button onClick={() => takeInsurance(true)} disabled={s.busy} className={`${btn} bg-[#d7b45b] text-[#10130e] hover:bg-[#efd58c]`}>
                Insurance (◈ {money(s.hands.reduce((a, h) => a + h.bet, 0) / 2)})
              </button>
              <button onClick={() => takeInsurance(false)} disabled={s.busy} className={`${btn} border border-white/20 hover:border-[#d7b45b] hover:text-[#e5c779]`}>
                No thanks
              </button>
            </div>
          ) : canBet ? (
            <div className="flex flex-col items-center gap-3">
              {outOfChips ? (
                <button onClick={refill} className={`${btn} bg-[#d7b45b] text-[#10130e] hover:bg-[#efd58c]`}>
                  Out of chips · take a ◈ {REFILL} refill
                </button>
              ) : (
                <button
                  onClick={deal}
                  disabled={s.bet < MIN_BET || stake > s.balance}
                  className={`${btn} bg-[#d7b45b] px-10 text-[#10130e] hover:bg-[#efd58c]`}
                >
                  {s.phase === "settled" ? "Deal again" : "Deal"} · ◈ {money(stake)}
                </button>
              )}
              <p className="text-[11px] text-white/30">Press Space to deal</p>
            </div>
          ) : (
            <div className="flex flex-wrap justify-center gap-3">
              <button onClick={hit} disabled={!playing} className={`${btn} bg-[#d7b45b] text-[#10130e] hover:bg-[#efd58c]`}>
                Hit <kbd className="ml-1 text-[10px] opacity-60">H</kbd>
              </button>
              <button onClick={stand} disabled={!playing} className={`${btn} border border-white/20 hover:border-[#d7b45b] hover:text-[#e5c779]`}>
                Stand <kbd className="ml-1 text-[10px] opacity-60">S</kbd>
              </button>
              <button onClick={double} disabled={!canDouble} className={`${btn} border border-[#d7b45b]/50 text-[#e5c779] hover:bg-[#d7b45b]/10`}>
                Double <kbd className="ml-1 text-[10px] opacity-60">D</kbd>
              </button>
              <button onClick={split} disabled={!canSplit} className={`${btn} border border-amber-400/40 bg-amber-400/10 text-amber-300 hover:bg-amber-400/20`}>
                Split <kbd className="ml-1 text-[10px] opacity-60">P</kbd>
              </button>
              <button onClick={surrender} disabled={!canSurrender} className={`${btn} border border-white/15 text-white/60 hover:border-red-300/50 hover:text-red-200`}>
                Surrender <kbd className="ml-1 text-[10px] opacity-60">R</kbd>
              </button>
            </div>
          )}
        </div>
      </section>

      {/* ================= SIDEBAR ================= */}
      <aside className="flex flex-col gap-5">
        {s.toast && (
          <div role="status" className="animate-pop fixed right-4 top-4 z-50 rounded-2xl border border-[#e5c779]/60 bg-[#151b15] px-5 py-3 text-sm font-semibold text-[#f5d778] shadow-2xl">
            ★ {s.toast}
          </div>
        )}
        <div className="rounded-3xl border border-[#c8a65b]/30 bg-[#151b15] p-6">
          <p className="text-xs text-white/50">Your chips</p>
          <p className="mt-1 text-3xl font-black text-[#e5c779]">
            ◈ <Counter value={s.balance} />
          </p>
        </div>

        <div className="rounded-3xl border border-[#c8a65b]/20 bg-[#101713] p-6">
          <div className="flex items-center justify-between">
            <h3 className="text-lg font-bold">Your bet</h3>
            <span className="text-lg font-bold text-[#e5c779]">◈ {money(s.bet)}</span>
          </div>
          <div className="mt-4 flex flex-wrap justify-center gap-2">
            {CHIPS.map((v) => (
              <button
                key={v}
                onClick={() => addChip(v)}
                disabled={!canBet || (s.bet + v) * spots + sides > s.balance}
                aria-label={`Add ${v} chips`}
                className={`flex h-14 w-14 items-center justify-center rounded-full border-4 border-dashed border-white/70 text-sm font-black text-white shadow-lg outline-none transition hover:-translate-y-0.5 focus-visible:ring-2 focus-visible:ring-[#e5c779] disabled:cursor-not-allowed disabled:opacity-30 disabled:hover:translate-y-0 ${CHIP_BG[v]}`}
              >
                {v}
              </button>
            ))}
          </div>
          <div className="mt-4 grid grid-cols-3 gap-2">
            <button onClick={() => setBetTo(s.lastBet)} disabled={!canBet} className={small}>Rebet</button>
            <button onClick={() => setBetTo(s.bet * 2)} disabled={!canBet || s.bet === 0} className={small}>×2</button>
            <button onClick={() => setBetTo(0)} disabled={!canBet || s.bet === 0} className={small}>Clear</button>
          </div>
          <div className="mt-4 space-y-2">
            <SideBet
              label="Perfect Pairs" hint="Your first 2 cards pair · up to 25:1" amount={s.ppBet}
              onChange={(d) => adjustSide("ppBet", d)}
              minusDisabled={!canBet || s.ppBet <= 0}
              plusDisabled={!canBet || s.ppBet >= 100 || stake + 10 > s.balance}
            />
            <SideBet
              label="21+3" hint="Your 2 cards + dealer's card · up to 100:1" amount={s.t3Bet}
              onChange={(d) => adjustSide("t3Bet", d)}
              minusDisabled={!canBet || s.t3Bet <= 0}
              plusDisabled={!canBet || s.t3Bet >= 100 || stake + 10 > s.balance}
            />
          </div>
        </div>

        <div className="rounded-3xl border border-white/[0.07] bg-[#101713] p-6">
          <h3 className="text-lg font-bold">Table rules</h3>
          <div className="mt-3 flex gap-2" role="group" aria-label="Number of decks">
            {[2, 6, 8].map((d) => (
              <button
                key={d}
                onClick={() => updateSettings({ decks: d })}
                disabled={!canBet}
                aria-pressed={s.settings.decks === d}
                className={`flex-1 rounded-full border py-2 text-xs outline-none transition focus-visible:ring-2 focus-visible:ring-[#e5c779] disabled:opacity-40 ${
                  s.settings.decks === d ? "border-[#d7b45b] bg-[#d7b45b]/10 text-[#e5c779]" : "border-white/10 text-white/60"
                }`}
              >
                {d} decks
              </button>
            ))}
          </div>
          <div className="mt-3 flex gap-2" role="group" aria-label="Hands per round">
            {[1, 2, 3].map((n) => (
              <button
                key={n}
                onClick={() => updateSettings({ spots: n })}
                disabled={!canBet}
                aria-pressed={spots === n}
                className={`flex-1 rounded-full border py-2 text-xs outline-none transition focus-visible:ring-2 focus-visible:ring-[#e5c779] disabled:opacity-40 ${
                  spots === n ? "border-[#d7b45b] bg-[#d7b45b]/10 text-[#e5c779]" : "border-white/10 text-white/60"
                }`}
              >
                {n} hand{n > 1 ? "s" : ""}
              </button>
            ))}
          </div>
          <div className="mt-3 flex gap-2" role="group" aria-label="Table theme">
            {(["green", "blue", "burgundy"] as Theme[]).map((th) => (
              <button
                key={th}
                onClick={() => updateSettings({ theme: th })}
                aria-pressed={s.settings.theme === th}
                className={`flex-1 rounded-full border py-2 text-xs capitalize outline-none transition focus-visible:ring-2 focus-visible:ring-[#e5c779] ${
                  s.settings.theme === th ? "border-[#d7b45b] bg-[#d7b45b]/10 text-[#e5c779]" : "border-white/10 text-white/60"
                }`}
              >
                {th}
              </button>
            ))}
          </div>
          <div className="mt-4 space-y-3">
            <Toggle label="Dealer hits soft 17" checked={s.settings.h17} disabled={!canBet} onChange={(v) => updateSettings({ h17: v })} />
            <Toggle label="Blackjack pays 6 to 5" checked={s.settings.bj65} disabled={!canBet} onChange={(v) => updateSettings({ bj65: v })} />
            <Toggle label="Sound" checked={!s.settings.muted} onChange={(v) => updateSettings({ muted: !v })} />
            <Toggle label="Strategy hints" checked={showHint} onChange={setShowHint} />
            <Toggle label="Card counting trainer" checked={showCount} onChange={setShowCount} />
          </div>
        </div>

        <div className="rounded-3xl border border-white/[0.07] bg-[#101713] p-6">
          <h3 className="text-lg font-bold">Session</h3>
          <dl className="mt-3 grid grid-cols-2 gap-y-2 text-xs text-white/50">
            <dt>Hands played</dt><dd className="text-right text-white">{s.stats.played}</dd>
            <dt>Won / lost / push</dt>
            <dd className="text-right text-white">{s.stats.won} / {s.stats.lost} / {s.stats.pushed}</dd>
            <dt>Win rate</dt><dd className="text-right text-white">{winRate}%</dd>
            <dt>Blackjacks</dt><dd className="text-right text-white">{s.stats.blackjacks}</dd>
            <dt>Profit</dt>
            <dd className={`text-right font-bold ${s.stats.profit >= 0 ? "text-emerald-300" : "text-red-300"}`}>
              {s.stats.profit >= 0 ? "+" : "−"}{money(Math.abs(s.stats.profit))}
            </dd>
          </dl>

          <div className="mt-4">
            <Spark data={s.trend} />
          </div>
          {s.history.length > 0 && (
            <div className="mt-3 flex flex-wrap gap-1.5" aria-label="Recent rounds">
              {s.history.map((r, i) => (
                <span
                  key={i}
                  title={`${r.net > 0 ? "+" : ""}${r.net}`}
                  className={`flex h-6 w-6 items-center justify-center rounded-full text-[10px] font-bold ${
                    r.bj ? "bg-[#e5c779] text-black" : r.net > 0 ? "bg-emerald-500/30 text-emerald-200" : r.net < 0 ? "bg-red-500/30 text-red-200" : "bg-white/15 text-white/70"
                  }`}
                >
                  {r.bj ? "BJ" : r.net > 0 ? "W" : r.net < 0 ? "L" : "P"}
                </span>
              ))}
            </div>
          )}

          <button onClick={resetAll} disabled={!canBet} className={`${small} mt-5 w-full`}>
            Reset chips and stats
          </button>
        </div>

        <div className="rounded-3xl border border-white/[0.07] bg-[#101713] p-6">
          <h3 className="text-lg font-bold">Coach</h3>
          <p className="mt-1 text-xs text-white/40">Each decision is graded against basic strategy.</p>
          <p className="mt-3 text-2xl font-black text-[#e5c779]">
            {s.stats.decisions ? Math.round((s.stats.correct / s.stats.decisions) * 100) : 0}%
          </p>
          <p className="text-[11px] text-white/40">{s.stats.correct} / {s.stats.decisions} correct decisions</p>
          <p className="mt-3 min-h-8 text-xs text-white/70" aria-live="polite">{s.coach ?? "Make a move to get feedback."}</p>
        </div>

        {showCount && (
          <div className="rounded-3xl border border-white/[0.07] bg-[#101713] p-6">
            <h3 className="text-lg font-bold">Hi-Lo count</h3>
            <dl className="mt-3 grid grid-cols-2 gap-y-2 text-xs text-white/50">
              <dt>Running count</dt>
              <dd className="text-right font-bold text-[#e5c779]">{running > 0 ? "+" : ""}{running}</dd>
              <dt>True count</dt>
              <dd className="text-right font-bold text-[#e5c779]">{trueCount > 0 ? "+" : ""}{trueCount.toFixed(1)}</dd>
              <dt>Decks left</dt>
              <dd className="text-right text-white">{decksLeft.toFixed(1)}</dd>
            </dl>
            <p className="mt-3 text-[11px] leading-4 text-white/35">
              2–6 count +1 · 7–9 count 0 · 10–A count −1. A high positive count means the shoe is rich in tens and aces.
            </p>
          </div>
        )}

        <div className="rounded-3xl border border-white/[0.07] bg-[#101713] p-6">
          <h3 className="text-lg font-bold">Records</h3>
          <dl className="mt-3 grid grid-cols-2 gap-y-2 text-xs text-white/50">
            <dt>Biggest win</dt><dd className="text-right text-white">◈ {money(s.records.biggestWin)}</dd>
            <dt>Best win streak</dt><dd className="text-right text-white">{s.records.bestStreak}</dd>
            <dt>Peak balance</dt><dd className="text-right text-white">◈ {money(s.records.peakBalance)}</dd>
          </dl>
          <ul className="mt-4 space-y-1.5 text-xs">
            {ACHIEVEMENTS.map((a) => {
              const done = s.unlocked.includes(a.id);
              return (
                <li key={a.id} className={done ? "text-[#e5c779]" : "text-white/30"}>
                  {done ? "★" : "☆"} {a.title} <span className="text-white/30">· {a.desc}</span>
                </li>
              );
            })}
          </ul>
        </div>

        <p className="px-3 text-center text-[11px] leading-5 text-white/25">
          Virtual chips only. No real money, deposits or cash-outs.
        </p>
      </aside>
    </div>
  );
