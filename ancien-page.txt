"use client";

import { useEffect, useRef, useState } from "react";

type Card = {
  rank: string;
  suit: string;
  value: number;
};


type CardFlight = {
  id: number;
  card: Card;
  destination: "dealer" | "player";
  startX: number;
  startY: number;
  endX: number;
  endY: number;
  rotating: number;
  animate: boolean;
};


const symbols = ["🍒", "🍋", "🍇", "💎", "🔔", "7️⃣"];
const suits = ["♠", "♥", "♦", "♣"];
const ranks = [
  { rank: "A", value: 11 },
  { rank: "2", value: 2 },
  { rank: "3", value: 3 },
  { rank: "4", value: 4 },
  { rank: "5", value: 5 },
  { rank: "6", value: 6 },
  { rank: "7", value: 7 },
  { rank: "8", value: 8 },
  { rank: "9", value: 9 },
  { rank: "10", value: 10 },
  { rank: "J", value: 10 },
  { rank: "Q", value: 10 },
  { rank: "K", value: 10 },
];

function makeDeck(): Card[] {
  return suits
    .flatMap((suit) =>
      ranks.map((card) => ({ ...card, suit }))
    )
    .sort(() => Math.random() - 0.5);
}

function scoreHand(hand: Card[]) {
  let score = hand.reduce((sum, card) => sum + card.value, 0);
  let aces = hand.filter((card) => card.rank === "A").length;

  while (score > 21 && aces > 0) {
    score -= 10;
    aces--;
  }

  return score;
}

export default function Home() {
  const [balance, setBalance] = useState(1000);
  const [bet, setBet] = useState(25);
  const [game, setGame] = useState<"slots" | "blackjack">("slots");
  
const shoeRef = useRef<HTMLDivElement>(null);
const dealerAreaRef = useRef<HTMLDivElement>(null);
const playerAreaRef = useRef<HTMLDivElement>(null);

const [cardFlight, setCardFlight] = useState<CardFlight | null>(null);
const flightIdRef = useRef(0);


  const [reels, setReels] = useState(["🍒", "💎", "🍋"]);
  const [slotMessage, setSlotMessage] = useState(
    "Place your bet and spin."
  );
  const [spinning, setSpinning] = useState(false);
  const [slotWin, setSlotWin] = useState(false);
  const [lastWin, setLastWin] = useState(0);

  const [deck, setDeck] = useState<Card[]>([]);
  const [player, setPlayer] = useState<Card[]>([]);
  const [dealer, setDealer] = useState<Card[]>([]);
  const [roundActive, setRoundActive] = useState(false);
  const [dealing, setDealing] = useState(false);
  const [visiblePlayerCards, setVisiblePlayerCards] = useState(0);
  const [visibleDealerCards, setVisibleDealerCards] = useState(0);
  const [revealDealerCard, setRevealDealerCard] = useState(false);
  const [activeBet, setActiveBet] = useState(25);
  const [canDouble, setCanDouble] = useState(false);
  const [splitMode, setSplitMode] = useState(false);
  const [splitHands, setSplitHands] = useState<Card[][]>([]);
  const [splitBets, setSplitBets] = useState<number[]>([]);
  const [activeHandIndex, setActiveHandIndex] = useState(0);
  const [finishedHands, setFinishedHands] = useState<boolean[]>([]);
  const [blackjackMessage, setBlackjackMessage] = useState(
    "Ready to play? Place your bet."
  );

function canSplitHand(hand: Card[]) {
  return (
    hand.length === 2 &&
    hand[0].rank === hand[1].rank
  );
}

  function spinSlots() {
    if (spinning || balance < bet) return;

    setSpinning(true);
    setSlotWin(false);
    setLastWin(0);
    setBalance((current) => current - bet);
    setSlotMessage("The reels are spinning...");

    const nextReels = Array.from(
      { length: 3 },
      () => symbols[Math.floor(Math.random() * symbols.length)]
    );

    window.setTimeout(() => {
      setReels(nextReels);

      const [a, b, c] = nextReels;
      let multiplier = 0;

      if (a === b && b === c) {
  multiplier = a === "7️⃣" ? 25 : a === "💎" ? 10 : 5;
} else if (a === b || b === c || a === c) {
  multiplier = 2;
}

      if (multiplier > 0) {
  const winnings = bet * multiplier;

  setBalance((current) => current + winnings);
  setSlotWin(true);
  setLastWin(winnings);

  setSlotMessage(
    a === "7️⃣" && b === "7️⃣" && c === "7️⃣"
      ? `👑 JACKPOT! +${winnings.toLocaleString("en-US")} chips!`
      : `✨ WINNER! +${winnings.toLocaleString("en-US")} chips`
  );
} else {
  setSlotWin(false);
  setLastWin(0);
  setSlotMessage("No win this time. Try your luck again!");
}

setSpinning(false);
    }, 650);
  }

  
function wait(ms: number) {
  return new Promise<void>((resolve) => {
    window.setTimeout(resolve, ms);
  });
}


  
async function finishRound(
  hand: Card[],
  dealerHand: Card[],
  currentBet: number,
  currentDeck: Card[]
) {
  setDealing(true);
  setCanDouble(false);

  let finalDealer = [...dealerHand];
  let remainingDeck = [...currentDeck];

  await wait(300);
setRoundActive(false);

await new Promise<void>((resolve) => {
  requestAnimationFrame(() => {
    requestAnimationFrame(() => resolve());
  });
});

setRevealDealerCard(true);

  while (
  scoreHand(finalDealer) < 17 &&
  remainingDeck.length > 0
) {
  await wait(200);

  const nextCard = remainingDeck.pop()!;

  await animateCardFlight(nextCard, "dealer");

  finalDealer.push(nextCard);

  setDealer([...finalDealer]);
  setDeck([...remainingDeck]);
  setVisibleDealerCards(finalDealer.length);
}

  const playerScore = scoreHand(hand);
  const dealerScore = scoreHand(finalDealer);

  setDealer(finalDealer);
  setDeck(remainingDeck);

  if (playerScore > 21) {
    setBlackjackMessage(
      `Bust! You scored ${playerScore}. The dealer wins.`
    );
  } else if (
    dealerScore > 21 ||
    playerScore > dealerScore
  ) {
    setBalance((current) => current + currentBet * 2);
    setBlackjackMessage(
      `You win! ${playerScore} vs ${dealerScore}. +${currentBet} chips profit.`
    );
  } else if (playerScore === dealerScore) {
    setBalance((current) => current + currentBet);
    setBlackjackMessage(
      `Push! Both scored ${playerScore}. Your bet is returned.`
    );
  } else {
    setBlackjackMessage(
      `Dealer wins: ${dealerScore} vs ${playerScore}. Better luck next time!`
    );
  }

  setDealing(false);
}


async function finishSplitRound(
  hands: Card[][],
  bets: number[],
  currentDeck: Card[],
  dealerHand: Card[]
) {
  setDealing(true);
  setCanDouble(false);
  setRoundActive(false);

  let finalDealer = [...dealerHand];
  let remainingDeck = [...currentDeck];

  setRevealDealerCard(true);

  const hasLiveHand = hands.some(
    (hand) => scoreHand(hand) <= 21
  );

  while (
  scoreHand(finalDealer) < 17 &&
  remainingDeck.length > 0
) {
  await wait(200);

  const nextCard = remainingDeck.pop()!;

  await animateCardFlight(nextCard, "dealer");

  finalDealer.push(nextCard);

  setDealer([...finalDealer]);
  setDeck([...remainingDeck]);
  setVisibleDealerCards(finalDealer.length);
}

  const dealerScore = scoreHand(finalDealer);
  let totalPayout = 0;
  const results: string[] = [];

  hands.forEach((hand, index) => {
    const playerScore = scoreHand(hand);
    const handBet = bets[index] ?? activeBet;

    if (playerScore > 21) {
      results.push(`Hand ${index + 1}: bust`);
    } else if (
      dealerScore > 21 ||
      playerScore > dealerScore
    ) {
      totalPayout += handBet * 2;
      results.push(`Hand ${index + 1}: win`);
    } else if (playerScore === dealerScore) {
      totalPayout += handBet;
      results.push(`Hand ${index + 1}: push`);
    } else {
      results.push(`Hand ${index + 1}: loss`);
    }
  });

  setBalance((current) => current + totalPayout);
  setDealer(finalDealer);
  setDeck(remainingDeck);
  setSplitHands(hands);
  setFinishedHands([true, true]);
  setVisibleDealerCards(finalDealer.length);

  setBlackjackMessage(
    `Split complete — ${results.join(" · ")}. Total returned: ${totalPayout} chips.`
  );

  setDealing(false);
}


  async function startBlackjack() {
  if (roundActive || dealing || balance < bet) return;

  const newDeck = makeDeck();
  const hand = [newDeck.pop()!, newDeck.pop()!];
  const dealerHand = [newDeck.pop()!, newDeck.pop()!];
  const remaining = [...newDeck];

  const playerNatural = scoreHand(hand) === 21;
  const dealerNatural = scoreHand(dealerHand) === 21;

  setBalance((current) => current - bet);
  setActiveBet(bet);
  setPlayer(hand);
  setDealer(dealerHand);
  setDeck(remaining);
  setSplitMode(false);
  setSplitHands([]);
  setSplitBets([]);
  setActiveHandIndex(0);
  setFinishedHands([]);
  setDealing(true);
  setRoundActive(false);
  setCanDouble(false);
  setVisiblePlayerCards(0);
  setVisibleDealerCards(0);
  setRevealDealerCard(false);

  await animateCardFlight(hand[0], "player");
  setVisiblePlayerCards(1);

  await animateCardFlight(dealerHand[0], "dealer");
  setVisibleDealerCards(1);

  await animateCardFlight(hand[1], "player");
  setVisiblePlayerCards(2);

  await animateCardFlight(dealerHand[1], "dealer");
  setVisibleDealerCards(2);

  setDealing(false);

  if (playerNatural || dealerNatural) {
    setRevealDealerCard(true);
    setRoundActive(false);

    if (playerNatural && dealerNatural) {
      setBalance((current) => current + bet);
      setBlackjackMessage(
        "Both have blackjack! Push — your bet is returned."
      );
    } else if (playerNatural) {
      setBalance((current) => current + bet * 2.5);
      setBlackjackMessage(
        `BLACKJACK! You win 3:2 — +${bet * 1.5} chips profit!`
      );
    } else {
      setBlackjackMessage(
        "Dealer has blackjack. You lose this round."
      );
    }

    return;
  }

  setRoundActive(true);
  setCanDouble(true);
  setBlackjackMessage(
    "Your move: hit, stand or double down."
  );
}


async function split() {
  if (
    !roundActive ||
    dealing ||
    splitMode ||
    !canSplitHand(player) ||
    balance < activeBet ||
    deck.length < 2
  ) {
    return;
  }

  setDealing(true);
  setBalance((current) => current - activeBet);

  const firstCard = player[0];
  const secondCard = player[1];
  const firstDraw = deck[deck.length - 1];
  const secondDraw = deck[deck.length - 2];
  const remaining = deck.slice(0, -2);

  const firstHand = [firstCard, firstDraw];
const secondHand = [secondCard, secondDraw];

setSplitMode(true);
setSplitHands([[firstCard], [secondCard]]);
setSplitBets([activeBet, activeBet]);
setFinishedHands([false, false]);
setActiveHandIndex(0);

setPlayer([firstCard]);
setDeck(remaining);

await animateCardFlight(firstDraw, "player");

setSplitHands([[firstCard, firstDraw], [secondCard]]);
setPlayer(firstHand);

await animateCardFlight(secondDraw, "player");

setSplitHands([firstHand, secondHand]);

  setBlackjackMessage(
    "Split! Play your first hand, then your second."
  );

  setCanDouble(true);
  setDealing(false);
}

  

async function hit() {
  if (!roundActive || dealing || deck.length === 0) return;

  setDealing(true);

  const card = deck[deck.length - 1];
  const remaining = deck.slice(0, -1);
  await animateCardFlight(card, "player");

  if (splitMode) {
    const hands = splitHands.map((hand) => [...hand]);
    hands[activeHandIndex].push(card);

    setSplitHands(hands);
    setDeck(remaining);
    setPlayer(hands[activeHandIndex]);
    setVisiblePlayerCards(hands[activeHandIndex].length);
    setCanDouble(false);

    await wait(350);

    const score = scoreHand(hands[activeHandIndex]);

    if (score >= 21) {
      const finished = [...finishedHands];
      finished[activeHandIndex] = true;
      setFinishedHands(finished);

      if (activeHandIndex === 0 && !finished[1]) {
        setActiveHandIndex(1);
        setPlayer(hands[1]);
        setVisiblePlayerCards(hands[1].length);
        setCanDouble(true);
        setBlackjackMessage("First hand finished. Play your second hand.");
        setDealing(false);
        return;
      }

      await finishSplitRound(
        hands,
        splitBets,
        remaining,
        dealer
      );
      return;
    }

    setDealing(false);
    return;
  }

  const nextHand = [...player, card];

  setPlayer(nextHand);
  setDeck(remaining);
  setVisiblePlayerCards(nextHand.length);

  const score = scoreHand(nextHand);

  if (score > 21) {
  setRevealDealerCard(true);
  await wait(750);

  setRoundActive(false);
  setBlackjackMessage(
    `Bust! You scored ${score}. The dealer wins.`
  );
  setCanDouble(false);
  setDealing(false);
} else if (score === 21) {
    await finishRound(nextHand, dealer, activeBet, remaining);
  } else {
    setCanDouble(false);
    setDealing(false);
  }
}



async function doubleDown() {
  const currentBet = splitMode
    ? splitBets[activeHandIndex]
    : activeBet;

  const currentHand = splitMode
    ? splitHands[activeHandIndex]
    : player;

  if (
    !roundActive ||
    dealing ||
    !canDouble ||
    balance < currentBet ||
    deck.length === 0 ||
    currentHand.length !== 2
  ) {
    return;
  }

  const card = deck[deck.length - 1];
  const remaining = deck.slice(0, -1);
  const nextHand = [...currentHand, card];

  setDealing(true);
  setBalance((current) => current - currentBet);
  setCanDouble(false);
  setDeck(remaining);

  await animateCardFlight(card, "player");

  if (splitMode) {
    const nextHands = splitHands.map((hand) => [...hand]);
    nextHands[activeHandIndex] = nextHand;

    const nextBets = [...splitBets];
    nextBets[activeHandIndex] = currentBet * 2;

    const nextFinished = [...finishedHands];
    nextFinished[activeHandIndex] = true;

    setSplitHands(nextHands);
    setSplitBets(nextBets);
    setFinishedHands(nextFinished);
    setPlayer(nextHand);
    setVisiblePlayerCards(nextHand.length);

    if (activeHandIndex === 0 && !nextFinished[1]) {
      setActiveHandIndex(1);
      setPlayer(nextHands[1]);
      setVisiblePlayerCards(nextHands[1].length);
      setCanDouble(
        nextHands[1].length === 2 &&
        balance - currentBet >= nextBets[1]
      );
      setBlackjackMessage(
        "First hand doubled. Play your second hand."
      );
      setDealing(false);
      return;
    }

    await finishSplitRound(
      nextHands,
      nextBets,
      remaining,
      dealer
    );
    return;
  }

  setActiveBet(currentBet * 2);
  setPlayer(nextHand);
  setVisiblePlayerCards(nextHand.length);

  if (scoreHand(nextHand) > 21) {
  setRevealDealerCard(true);
  await wait(750);

  setRoundActive(false);
  setBlackjackMessage(
    `Bust! You scored ${scoreHand(nextHand)}. The dealer wins.`
  );
  setCanDouble(false);
  setDealing(false);
}

  await finishRound(
    nextHand,
    dealer,
    currentBet * 2,
    remaining
  );
}


async function stand() {
  if (!roundActive || dealing) return;

  if (splitMode) {
    const finished = [...finishedHands];
    finished[activeHandIndex] = true;
    setFinishedHands(finished);
    setCanDouble(false);
    setDealing(true);

    await wait(250);

    if (activeHandIndex === 0 && !finished[1]) {
      setActiveHandIndex(1);
      setPlayer(splitHands[1]);
      setVisiblePlayerCards(splitHands[1].length);
      setCanDouble(true);
      setBlackjackMessage("First hand finished. Play your second hand.");
      setDealing(false);
      return;
    }

    await finishSplitRound(
      splitHands,
      splitBets,
      deck,
      dealer
    );
    return;
  }

  await finishRound(player, dealer, activeBet, deck);
}


  
function resetBalance() {
  if (roundActive || dealing || spinning) return;

  setBalance(1000);
  setBet(25);
  setSlotMessage("Place your bet and spin.");
  setBlackjackMessage("Ready to play? Place your bet.");

  setSplitMode(false);
  setSplitHands([]);
  setSplitBets([]);
  setActiveHandIndex(0);
  setFinishedHands([]);
  setCanDouble(false);
}


  const money = (amount: number) =>
    amount.toLocaleString("en-US");
  
async function animateCardFlight(
  card: Card,
  destination: "dealer" | "player"
) {
  const shoe = shoeRef.current;
  const target =
    destination === "dealer"
      ? dealerAreaRef.current
      : playerAreaRef.current;

  if (!shoe || !target) return;

  const shoeRect = shoe.getBoundingClientRect();
  const targetRect = target.getBoundingClientRect();

  const startX = shoeRect.left + shoeRect.width / 2;
  const startY = shoeRect.top + shoeRect.height / 2;

  const endX =
    targetRect.left +
    targetRect.width / 2 +
    (Math.random() - 0.5) * 50;

  const endY = targetRect.top + targetRect.height / 2;
  const id = ++flightIdRef.current;

  setCardFlight({
  id,
  card,
  destination,
  startX,
  startY,
  endX,
  endY,
  rotating: (Math.random() - 0.5) * 12,
  animate: false,
});

  await new Promise<void>((resolve) => {
    requestAnimationFrame(() => {
      requestAnimationFrame(() => {
        setCardFlight((current) =>
          current?.id === id
            ? { ...current, animate: true }
            : current
        );
        resolve();
      });
    });
  });

  await wait(650);

  setCardFlight((current) =>
    current?.id === id ? null : current
  );

  // Let React render the removal before the next card flies.
  await new Promise<void>((resolve) => {
    requestAnimationFrame(() => resolve());
  });
}

  className={`rounded-3xl border border-[#c8a65b]/30 bg-[radial-gradient(ellipse_at_center,#254336_0%,#12271e_48%,#0a120e_100%)] p-5 sm:p-9 transition-all duration-300 ${
    spinning
      ? "animate-pulse"
      : slotWin
      ? "ring-2 ring-[#e5c779] shadow-[0_0_45px_#e5c77955]"
      : ""
  }`}
>
                    <div className="mb-5 text-center text-[10px] tracking-[0.35em] text-[#d7b45b]/80">
                      ✦ &nbsp; SPIN FOR GLORY &nbsp; ✦
                    </div>

                    <div className="mx-auto grid max-w-md grid-cols-3 gap-2 sm:gap-4">
                      {reels.map((symbol, index) => (
                        <div
                          key={index}
                          className="flex aspect-[0.9] items-center justify-center rounded-2xl border-2 border-[#c8a65b]/50 bg-[#f5eedc] text-5xl shadow-[inset_0_0_20px_#c8a65b33] sm:text-7xl"
                        >
                          <span
  className={
    spinning
      ? "animate-bounce"
      : slotWin
      ? "drop-shadow-[0_0_18px_#e5c779]"
      : ""
  }
>
  {symbol}
</span>
                        </div>
                      ))}
                    </div>

                    <div className="mt-5 text-center text-xs text-[#e5c779]">
                      {slotMessage}
                    </div>
                  </div>

                  {lastWin > 0 && (
  <div
    className={`mt-4 text-center ${
      reels.every((symbol) => symbol === "7️⃣")
        ? "animate-bounce text-2xl font-black tracking-widest text-[#f5d778]"
        : "text-lg font-bold text-[#e5c779]"
    }`}
  >
    {reels.every((symbol) => symbol === "7️⃣")
      ? "👑 ROYAL JACKPOT 👑"
      : "WINNINGS"}
    <div className="mt-1 text-sm">
      +{lastWin.toLocaleString("en-US")} virtual chips
    </div>
  </div>
)}

                  <div className="mt-6 grid grid-cols-3 gap-2 rounded-2xl border border-white/[0.07] bg-black/20 p-4 text-center">
                    <div>
                      <div className="text-xl">🍒</div>
                      <div className="mt-1 text-xs text-white/60">
                        Pair
                      </div>
                      <div className="text-sm font-bold text-[#e5c779]">
                        2×
                      </div>
                    </div>
                    <div>
                      <div className="text-xl">💎</div>
                      <div className="mt-1 text-xs text-white/60">
                        Triple diamonds
                      </div>
                      <div className="text-sm font-bold text-[#e5c779]">
                        10×
                      </div>
                    </div>
                    <div>
                      <div className="text-xl">7️⃣</div>
                      <div className="mt-1 text-xs text-white/60">
                        Triple sevens
                      </div>
                      <div className="text-sm font-bold text-[#e5c779]">
                        25×
                      </div>
                    </div>
                  </div>
                </div>
              </>
            ) : (
              <>
                <div className="flex items-center justify-between border-b border-white/[0.07] px-6 py-5">
                  <div>
                    <p className="text-[10px] tracking-[0.3em] text-[#d7b45b]">
                      CLASSIC TABLE
                    </p>
                    <h3 className="mt-1 text-xl font-bold">
                      Blackjack
                    </h3>
                  </div>
                  <span className="rounded-full border border-[#d7b45b]/30 px-3 py-1 text-[10px] text-[#e5c779]">
                    21
                  </span>
                </div>

                <div className="bg-[radial-gradient(ellipse_at_center,#20543c_0%,#103524_60%,#09170f_100%)] p-5 sm:p-9">
                  
<div className="mb-6 flex items-center justify-end gap-4">
  <div className="text-right">
    <p className="text-[9px] font-semibold tracking-[0.3em] text-[#d7b45b]/70">
      ROYAL ACE
    </p>
    <p className="mt-1 text-xs text-white/45">
      Card shoe
    </p>
  </div>

  <div
    ref={shoeRef}
    className="relative flex h-24 w-20 shrink-0 items-center justify-center rounded-xl border border-[#d7b45b]/60 bg-[#101d18] shadow-[0_8px_25px_#0008]"
  >
    <div className="absolute left-2 top-1 h-[78px] w-[52px] rotate-[-9deg] rounded-lg border border-[#d7b45b]/40 bg-[#1d4535]" />
    <div className="absolute left-3 top-1 h-[78px] w-[52px] rotate-[-4deg] rounded-lg border border-[#d7b45b]/50 bg-[#183a2d]" />
    <div className="relative flex h-[78px] w-[52px] items-center justify-center rounded-lg border border-[#d7b45b] bg-[#142d27] text-2xl text-[#e5c779] shadow-lg">
      ♠
    </div>
  </div>
</div>

                  <div className="text-center text-[10px] tracking-[0.3em] text-white/50">
                    THE DEALER
                  </div>
                  <div
  ref={dealerAreaRef}
  className="mt-4 flex min-h-32 justify-center gap-2"
>
                    
{dealer
  .slice(0, dealing ? visibleDealerCards : dealer.length)
  .map((card, index) => (
    <PlayingCard
  key={`${index}-${card.rank}-${card.suit}`}
  card={card}
  hidden={index === 1 && !revealDealerCard}
  animateFlip={index === 1}
/>
  ))}

                  </div>
                  <div className="mt-3 text-center text-sm text-white/60">
  {dealer.length === 0
    ? "Waiting for the deal"
    : !revealDealerCard
      ? "Hidden card"
      : `Score: ${scoreHand(dealer)}`}
</div>

                  <div className="my-7 border-t border-white/15" />

                  <div className="text-center text-[10px] tracking-[0.3em] text-white/50">
                    YOUR HAND
                  </div>
                  <div
  ref={playerAreaRef}
  className="mt-4 flex min-h-32 justify-center gap-2"
>
                    

{splitMode ? (
  <div className="grid w-full grid-cols-2 gap-4">
    {splitHands.map((hand, handIndex) => (
      <div
        key={handIndex}
        className={`rounded-2xl border p-3 transition ${
          activeHandIndex === handIndex && roundActive
            ? "border-amber-400/70 bg-amber-400/10"
            : "border-white/10 bg-black/10"
        }`}
      >
        <div className="mb-3 flex items-center justify-between gap-2">
          <span className="text-sm font-semibold text-white">
            Hand {handIndex + 1}
          </span>
          <span className="text-xs text-white/60">
            {scoreHand(hand)} pts
          </span>
        </div>

        <div className="flex flex-wrap justify-center gap-2">
          {hand.map((card, index) => (
            <PlayingCard key={index} card={card} />
          ))}
        </div>

        <p className="mt-3 text-center text-xs text-white/50">
          Bet: {splitBets[handIndex]} chips
          {finishedHands[handIndex] ? " · Finished" : ""}
        </p>
      </div>
    ))}
  </div>
) : (
  player
    .slice(0, dealing ? visiblePlayerCards : player.length)
    .map((card, index) => (
      <PlayingCard key={index} card={card} />
    ))
)}


                  </div>
                  <div className="mt-3 text-center text-sm font-bold text-[#e5c779]">
  {visiblePlayerCards > 0
    ? `Score: ${scoreHand(
        splitMode
          ? splitHands[activeHandIndex] ?? []
          : player.slice(0, visiblePlayerCards)
      )}`
    : "Your cards will appear here"}
</div>
                </div>

                <div className="p-5 sm:p-7">
                  <p className="mb-5 text-center text-sm leading-6 text-white/60">
                    {blackjackMessage}
                  </p>

                  <div className="flex flex-wrap justify-center gap-3">
  {!roundActive ? (
    <button
      onClick={startBlackjack}
      disabled={balance < bet || dealing}
      className="rounded-full bg-[#d7b45b] px-7 py-3 text-sm font-bold text-[#10130e] transition hover:bg-[#efd58c] disabled:opacity-30"
    >
      Deal cards
    </button>
  ) : (
    <>
      <button
        onClick={hit}
        disabled={dealing}
        className="rounded-full bg-[#d7b45b] px-7 py-3 text-sm font-bold text-[#10130e] transition hover:bg-[#efd58c]"
      >
        Hit
      </button>

      <button
        onClick={stand}
        disabled={dealing}
        className="rounded-full border border-white/20 px-7 py-3 text-sm font-bold transition hover:border-[#d7b45b] hover:text-[#e5c779]"
      >
        Stand
      </button>

      
{!splitMode &&
  roundActive &&
  canSplitHand(player) &&
  balance >= activeBet && (
    <button
      onClick={split}
      disabled={dealing}
      className="rounded-xl border border-amber-400/40 bg-amber-400/10 px-5 py-3 font-semibold text-amber-300 transition hover:bg-amber-400/20 disabled:cursor-not-allowed disabled:opacity-40"
    >
      Split
    </button>
  )}


      {canDouble && (
        <button
          onClick={doubleDown}
          disabled={balance < activeBet || dealing}
          className="rounded-full border border-[#d7b45b]/50 px-7 py-3 text-sm font-bold text-[#e5c779] transition hover:bg-[#d7b45b]/10 disabled:opacity-30"
        >
          Double Down
        </button>
      )}
    </>
  )}
</div>
                </div>
              </>
            )}
          </section>

          <aside className="flex flex-col gap-5">
            <div className="rounded-3xl border border-[#c8a65b]/20 bg-[#101713] p-6">
              <p className="text-[10px] tracking-[0.3em] text-[#d7b45b]">
                TABLE SETTINGS
              </p>
              <h3 className="mt-2 text-xl font-bold">Your wager</h3>
              <p className="mt-2 text-xs leading-5 text-white/40">
                Choose your stake before playing. All chips are
                virtual.
              </p>

              <div className="mt-6 rounded-2xl border border-white/10 bg-black/20 p-4">
                <div className="flex items-center justify-between text-xs text-white/50">
                  <span>Bet per round</span>
                  <span className="text-[#e5c779]">
                    ◈ {bet}
                  </span>
                </div>
                <div className="mt-4 grid grid-cols-2 gap-2">
                  {[10, 25, 50, 100].map((amount) => (
                    <button
                      key={amount}
                      onClick={() => setBet(amount)}
                      className={`rounded-xl border py-3 text-sm font-semibold transition ${
                        bet === amount
                          ? "border-[#d7b45b] bg-[#d7b45b]/10 text-[#e5c779]"
                          : "border-white/10 text-white/60 hover:border-white/30"
                      }`}
                    >
                      ◈ {amount}
                    </button>
                  ))}
                </div>
              </div>

              {game === "slots" && (
                <button
                  onClick={spinSlots}
                  disabled={spinning || balance < bet}
                  className="mt-5 w-full rounded-full bg-[#d7b45b] py-4 text-sm font-black tracking-[0.15em] text-[#10130e] transition hover:bg-[#efd58c] disabled:cursor-not-allowed disabled:opacity-30"
                >
                  {spinning ? "SPINNING..." : "SPIN REELS  ↗"}
                </button>
              )}

              {balance < bet && (
                <p className="mt-3 text-center text-xs text-red-300">
                  Not enough chips. Reset your balance to continue.
                </p>
              )}
            </div>

            <div className="rounded-3xl border border-white/[0.07] bg-[#101713] p-6">
              <p className="text-[10px] tracking-[0.3em] text-[#d7b45b]">
                HOUSE RULES
              </p>
              <h3 className="mt-2 text-lg font-bold">
                Play your way.
              </h3>
              <ul className="mt-4 space-y-3 text-xs leading-5 text-white/50">
                <li>✦ Start with 1,000 virtual chips.</li>
                <li>✦ Matching slot symbols pay out.</li>
                <li>✦ Blackjack aims for 21 without going over.</li>
                <li>✦ Beat the dealer to win your stake plus an equal payout.</li>
                <li>✦ No real money, deposits or cash-outs.</li>
              </ul>
            </div>

            <div className="px-3 text-center text-[10px] leading-5 tracking-[0.12em] text-white/25">
              ROYAL ACE · PRIVATE VIRTUAL GAME ROOM
              <br />
              FOR ENTERTAINMENT ONLY
            </div>
          </aside>
        </div>
            </section>

{cardFlight && (
  <div
    key={cardFlight.id}
    className="pointer-events-none fixed z-[100] h-28 w-[76px] sm:h-32 sm:w-[88px]"
    style={{
      left: cardFlight.animate
        ? cardFlight.endX
        : cardFlight.startX,
      top: cardFlight.animate
        ? cardFlight.endY
        : cardFlight.startY,
      transform: `translate(-50%, -50%) rotate(${
        cardFlight.animate ? cardFlight.rotating : -8
      }deg) scale(${cardFlight.animate ? 1 : 0.68})`,
      transition:
        "left 650ms cubic-bezier(0.22, 1, 0.36, 1), top 650ms cubic-bezier(0.22, 1, 0.36, 1), transform 650ms cubic-bezier(0.22, 1, 0.36, 1)",
    }}
  >
    <div
      className="relative h-full w-full"
      style={{
  perspective: "1000px",
  transformStyle: "preserve-3d",
  transform: `rotateY(${
    cardFlight.destination === "player" ? 180 : 0
  }deg)`,
}}
    >
      <div
        className="absolute inset-0 flex items-center justify-center rounded-xl border border-[#b9974d]/50 bg-[#142d27] text-3xl text-[#d7b45b] shadow-[0_12px_40px_#0009]"
        style={{
          backfaceVisibility: "hidden",
          WebkitBackfaceVisibility: "hidden",
        }}
      >
        ♠
      </div>

      <div
        className="absolute inset-0 flex flex-col justify-between rounded-xl bg-[#f8f5ec] p-2 shadow-[0_12px_40px_#0009]"
        style={{
          transform: "rotateY(180deg)",
          backfaceVisibility: "hidden",
          WebkitBackfaceVisibility: "hidden",
        }}
      >
        <span
          className={`text-lg font-bold ${
            cardFlight.card.suit === "♥" ||
            cardFlight.card.suit === "♦"
              ? "text-red-600"
              : "text-slate-950"
          }`}
        >
          {cardFlight.card.rank}
          <br />
          {cardFlight.card.suit}
        </span>

        <span
          className={`self-center text-3xl ${
            cardFlight.card.suit === "♥" ||
            cardFlight.card.suit === "♦"
              ? "text-red-600"
              : "text-slate-950"
          }`}
        >
          {cardFlight.card.suit}
        </span>
      </div>
    </div>
  </div>
)}
    </main>
  );
}
